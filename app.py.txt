import streamlit as st
import pandas as pd

st.set_page_config(page_title="Bond AI Screener", layout="wide")

@st.cache_data
def load_data():
    return pd.read_excel("data.xlsx")

df = load_data()

# Sorting
sort_by = st.sidebar.selectbox(
    "Sort By",
    ["SCORE", "SPREAD (bps)", "YTM (%)"]
)

df = df.sort_values(by=sort_by, ascending=False)

# Valuation
def valuation(row):
    if row["SPREAD (bps)"] > 80:
        return "CHEAP 🟢"
    elif row["SPREAD (bps)"] > 60:
        return "FAIR 🟡"
    else:
        return "EXPENSIVE 🔴"

df["VALUATION"] = df.apply(valuation, axis=1)

st.title("📊 Bond AI Screener")

st.subheader("Top Opportunities")
st.dataframe(
    df[[
        "SECURITY NAME",
        "PRICE",
        "YTM (%)",
        "SPREAD (bps)",
        "DELTA (bps)",
        "SCORE",
        "VALUATION"
    ]].head(5),
    use_container_width=True
)

st.subheader("Full Market View")
st.dataframe(
    df[[
        "SECURITY NAME",
        "PRICE",
        "YTM (%)",
        "SPREAD (bps)",
        "DELTA (bps)",
        "VOLUME",
        "SCORE",
        "VALUATION"
    ]],
    use_container_width=True
)

st.subheader("Spread Comparison")
st.bar_chart(df.set_index("SECURITY NAME")["SPREAD (bps)"])

best = df.iloc[0]

st.subheader("Best Trade")
st.success(f"""
{best['SECURITY NAME']}

Price: {best['PRICE']}
YTM: {best['YTM (%)']}%
Spread: {best['SPREAD (bps)']} bps
""")
